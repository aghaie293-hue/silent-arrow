        const SECURE_HASH = "5f64f5aa23cc4318a925608861b31455e409bee91daafde093c538897dae9241";

        async function sha256(str) {
            const buffer = new TextEncoder().encode(str);
            const hashBuffer = await crypto.subtle.digest("SHA-256", buffer);
            return Array.from(new Uint8Array(hashBuffer)).map(b => b.toString(16).padStart(2, '0')).join('');
        }

        async function checkAccess() {
            const inputVal = document.getElementById('passInput').value;
            const inputHash = await sha256(inputVal);
            if (inputHash === SECURE_HASH) {
                localStorage.setItem("sa_auth_token", "authorized_user");
                document.getElementById('authOverlay').style.display = 'none';
            } else {
                document.getElementById('authError').innerText = "رمز عبور نادرست است!";
            }
        }

        const median = arr => { const a = [...arr].sort((x, y) => x - y); return a[Math.floor(a.length / 2)]; };

        // ثبت از دوربین: ۳ ثانیه آماده‌باش، سپس ۳۰ نمونه در ~۳ ثانیه؛ میانه‌ی نمونه‌ها استفاده می‌شود
        async function captureCalibration() {
            if (calibBusy) return;
            if (!currentMetrics || currentMetrics.drawElbow === 0) {
                $('calibMsg').innerText = 'بدنی شناسایی نشده است. دوربین را روشن کنید و کامل در کادر بایستید.';
                return;
            }
            calibBusy = true;
            const btn = $('calibCaptureBtn');
            btn.disabled = true;
            for (let i = 3; i > 0; i--) {
                $('calibMsg').innerText = `آماده شوید و در کشش کامل ثابت بمانید... ${i}`;
                await new Promise(r => setTimeout(r, 1000));
            }
            const S = { elbow: [], bow: [], head: [], shoulder: [] };
            let tries = 0;
            while (S.elbow.length < 30 && tries++ < 60) {
                $('calibMsg').innerText = `در حال ثبت... ${S.elbow.length}/30 (ثابت بمانید)`;
                const m = currentMetrics;
                if (m && m.drawElbow !== 0) {
                    S.elbow.push(m.drawElbow); S.bow.push(m.bowArm);
                    S.head.push(m.headTilt); S.shoulder.push(m.shoulderSlope);
                }
                await new Promise(r => setTimeout(r, 100));
            }
            calibBusy = false;
            btn.disabled = false;

            if (S.elbow.length < 20) {
                $('calibMsg').innerText = 'بدن در حین ثبت بارها از کادر خارج شد. دوباره تلاش کنید.';
                return;
            }
            const spread = Math.max(...S.elbow) - Math.min(...S.elbow);
            const spreadBow = Math.max(...S.bow) - Math.min(...S.bow);
            if (spread > 15 || spreadBow > 15) {
                $('calibMsg').innerText = `حرکت زیاد بود (نوسان آرنج ${spread}°، دست کمان ${spreadBow}°). در حالت ثابت دوباره ثبت کنید. اعداد تغییر نکردند.`;
                return;
            }
            const head = median(S.head), shoulder = median(S.shoulder);
            fillCalibForm({
                drawElbowTarget: median(S.elbow),
                bowArmTarget: median(S.bow),
                toleranceElbow: Math.max(6, Math.ceil(spread / 2) + 4),
                toleranceBowArm: Math.max(6, Math.ceil(spreadBow / 2) + 4),
                headTiltMax: Math.min(6, Math.max(4, head + 1)),
                shoulderSlopeMax: Math.min(6, Math.max(4, shoulder + 1))
            });
            let msg = `ثبت شد (${S.elbow.length} نمونه). اعداد را بررسی و «ذخیره» را بزنید.`;
            if (head > 5 || shoulder > 5) msg += ` هشدار: کجی سر (${head}°) یا شانه (${shoulder}°) در این کشش زیاد بود؛ حد مجاز به ۶° محدود شد تا فرم بد معیار نشود.`;
            $('calibMsg').innerText = msg;
        }

        function saveCalibrationForm() {
            const spec = [
                ['cfgElbow', 'drawElbowTarget', 0, 180],
                ['cfgTolElbow', 'toleranceElbow', 1, 30],
                ['cfgBowArm', 'bowArmTarget', 90, 180],
                ['cfgTolBow', 'toleranceBowArm', 1, 30],
                ['cfgHead', 'headTiltMax', 1, 15],
                ['cfgShoulder', 'shoulderSlopeMax', 1, 15]
            ];
            const next = { ...personalProfile };
            for (const [id, key, lo, hi] of spec) {
                const v = Number($(id).value);

            next.version = 2;

            applyCalibLabel();
            $('calibMsg').innerText = '';
            calibBusy = false;
            $('calibModal').classList.remove('open');
        }

        // مختصات نرمال‌شده (۰ تا ۱) روی x و y مقیاس متفاوت دارند؛ قبل از زاویه به پیکسل تبدیل می‌شوند
        function calculateAngle(A, B, C) {
            const W = canvasElement.width, H = canvasElement.height;
            const radians = Math.atan2((C.y - B.y) * H, (C.x - B.x) * W) - Math.atan2((A.y - B.y) * H, (A.x - B.x) * W);
            let angle = Math.abs(radians * 180.0 / Math.PI);
            if (angle > 180.0) angle = 360 - angle;
            return Math.round(angle);
        }

        // شیب خط بین دو نقطه نسبت به افق، به درجه (در فضای پیکسل)
        function lineTiltDeg(P, Q) {
            const W = canvasElement.width, H = canvasElement.height;
            let d = Math.atan2((Q.y - P.y) * H, (Q.x - P.x) * W) * 180 / Math.PI;
            if (d > 90) d -= 180;
            if (d < -90) d += 180;
            return Math.abs(Math.round(d));
        }

        // سیستم تحلیل عیب‌یابی و تمرین اصلاحی
        function diagnoseAndRecommend(metrics, profile) {
            const drawDiff = Math.abs(metrics.drawElbow - profile.drawElbowTarget);
            const bowDiff = Math.abs(metrics.bowArm - profile.bowArmTarget);

            let diagnosis = "ایده‌آل (Optimal)";
            let drill = "فرم در دامنه استاندارد است؛ ادامه شلیک بدون تغییر";
            let priority = "سبک";

            if (metrics.headTilt > profile.headTiltMax) {
                diagnosis = "انحراف و ناپایداری سر در انکر";
                drill = "تمرین ثبات سر مقابل آینه با کش تمرینی (CLBWH)، ۳ ست ۳۰ ثانیه نگهداشت انکر بدون رهایی";
                priority = "بالا";
            } else if (metrics.shoulderSlope > profile.shoulderSlopeMax) {
                diagnosis = "بالا آمدن شانه یا عدم تقارن ترقوه";
                drill = "تمرین فعال‌سازی رومبوئید و پایین‌آوردن کتف (Scapular Depression) با کش سبک، ۳ ست ۱۲ تکرار";
                priority = "بالا";
            } else if (drawDiff > profile.toleranceElbow) {
                diagnosis = `انحراف آرنج کشش (${drawDiff}° اختلاف)`;
                drill = "تمرین چرخش به خط نیرو و امتداد خط دراو با کش، ۳ ست ۱۰ تکرار با تمرکز بر آرنج پشتی";
                priority = "متوسط";
            } else if (bowDiff > profile.toleranceBowArm) {
                diagnosis = `افت زاویه دست کمان (${bowDiff}° اختلاف)`;
                drill = "تمرین هل دادن به سمت هدف (Push-Pull Balance) و قفل نرم مفصل دست کمان، ۳ ست ۲۰ ثانیه";
                priority = "متوسط";
            }

            return { diagnosis, drill, priority, drawDiff, bowDiff };
        }

        function onResults(results) {
            loadingStatus.style.display = 'none';

            // اندازه canvas دقیقاً برابر اندازه واقعی فریم دوربین (جلوگیری از کشیدگی و عدم انطباق)
            const imgW = results.image.videoWidth || results.image.width;
            const imgH = results.image.videoHeight || results.image.height;
            if (imgW && imgH && (canvasElement.width !== imgW || canvasElement.height !== imgH)) {
                canvasElement.width = imgW;
                canvasElement.height = imgH;
                document.querySelector('.camera-container').style.aspectRatio = `${imgW} / ${imgH}`;
            }
            const W = canvasElement.width, H = canvasElement.height;
            const mirrored = (currentFacingMode === "user");

            canvasCtx.save();
            canvasCtx.clearRect(0, 0, W, H);
            // آینه‌سازی فقط برای دوربین جلو؛ تصویر و اسکلت با هم آینه می‌شوند
            if (mirrored) { canvasCtx.translate(W, 0); canvasCtx.scale(-1, 1); }
            canvasCtx.drawImage(results.image, 0, 0, W, H);

            if (results.poseLandmarks) {
                drawConnectors(canvasCtx, results.poseLandmarks, POSE_CONNECTIONS, {color: '#00e676', lineWidth: 3});
                drawLandmarks(canvasCtx, results.poseLandmarks, {color: '#ff5252', lineWidth: 1, radius: 4});
                canvasCtx.setTransform(1, 0, 0, 1, 0, 0); // متن‌ها آینه نشوند

                const lm = results.poseLandmarks;

                // گام ۱: اگر اندام‌های کلیدی خارج از کادر یا نامطمئن باشند، زاویه ثبت نمی‌شود
                const keyIdx = [7, 8, 11, 12, 13, 14, 15, 16];
                const visOK = keyIdx.every(i => (lm[i].visibility ?? 1) >= 0.6);
                if (!visOK) {
                    document.getElementById('headStabilityVal').innerText = 'خارج از کادر';
                    document.getElementById('shoulderAlignVal').innerText = 'خارج از کادر';
                    currentMetrics = null;
                    canvasCtx.restore();
                    return;
                }

                const rightElbowAngle = calculateAngle(lm[12], lm[14], lm[16]);
                const leftElbowAngle = calculateAngle(lm[11], lm[13], lm[15]);

                const leftEar = lm[7];
                const rightEar = lm[8];
                const headTilt = lineTiltDeg(leftEar, rightEar);
                const shoulderSlope = lineTiltDeg(lm[11], lm[12]);

                const headStatus = (headTilt <= personalProfile.headTiltMax) ? "پایدار" : "انحراف سر";
                const shoulderStatus = (shoulderSlope <= personalProfile.shoulderSlopeMax) ? "تراز" : "عدم تقارن";

                currentMetrics = {
                    drawElbow: rightElbowAngle,
                    bowArm: leftElbowAngle,
                    headTilt: headTilt,
                    shoulderSlope: shoulderSlope,
                    headStatus: headStatus,
                    shoulderStatus: shoulderStatus
                };

                document.getElementById('drawElbowVal').innerText = rightElbowAngle + '°';
                document.getElementById('bowArmVal').innerText = leftElbowAngle + '°';
                document.getElementById('headStabilityVal').innerText = headStatus;
                document.getElementById('shoulderAlignVal').innerText = shoulderStatus;

                // نمایش زنده روی تصویر
                canvasCtx.fillStyle = "#ffff00";
                canvasCtx.font = "bold 18px Arial";
                const px = (l) => (mirrored ? 1 - l.x : l.x) * W + 10;
                canvasCtx.fillText(rightElbowAngle + "°", px(lm[14]), lm[14].y * H);
                canvasCtx.fillText(leftElbowAngle + "°", px(lm[13]), lm[13].y * H);
            }
            canvasCtx.restore();
        }

        async function initSystem() {
            document.getElementById('startBtn').style.display = 'none';
            document.getElementById('flipBtn').style.display = 'inline-block';
            document.getElementById('calibBtn').style.display = 'inline-block';
            document.getElementById('snapBtn').style.display = 'inline-block';
            document.getElementById('logBtn').style.display = 'inline-block';
            document.getElementById('exportBtn').style.display = 'inline-block';
            loadingStatus.style.display = 'block';

            // گام ۱: اندازه canvas قبل از شروع دوربین تنظیم می‌شود
            canvasElement.width = 640;
            canvasElement.height = 480;
            document.querySelector('.camera-container').style.aspectRatio = '4 / 3';

            // گام ۱: Pose فقط یک‌بار ساخته می‌شود (جلوگیری از نشت حافظه/WebGL)
            if (!poseDetector) {
                poseDetector = new Pose({
                    locateFile: (file) => `https://cdn.jsdelivr.net/npm/@mediapipe/pose@0.5.1675469404/${file}`
                });
                poseDetector.setOptions({
                    modelComplexity: 1,
                    smoothLandmarks: true,
                    minDetectionConfidence: 0.5,
                    minTrackingConfidence: 0.5
                });
                poseDetector.onResults(onResults);
            }

            if (camera) { await camera.stop(); camera = null; }

            camera = new Camera(videoElement, {
                onFrame: async () => {
                    await poseDetector.send({image: videoElement});
                },
                width: 640,
                height: 480,
                facingMode: currentFacingMode
            });

            await camera.start();
            loadingStatus.style.display = 'none';
        }

        async function toggleCamera() {
            currentFacingMode = (currentFacingMode === "user") ? "environment" : "user";
            await initSystem();
        }

        function takeSnapshot() {
            const link = document.createElement('a');
            const now = new Date().toISOString().replace(/[:.]/g, "-");
            link.download = `SilentArrows_Snapshot_${now}.png`;
            link.href = canvasElement.toDataURL("image/png");
            link.click();
        }

        function logCurrentShot() {
            if (currentMetrics.drawElbow === 0) {
                alert("بدنی در کادر شناسایی نشده است.");
                return;
            }

            const shotNumber = sessionLogs.length + 1;
            const timeStr = new Date().toLocaleTimeString('fa-IR');
            const analysis = diagnoseAndRecommend(currentMetrics, personalProfile);

            sessionLogs.push({
                shot: shotNumber,
                time: timeStr,
                drawElbow: currentMetrics.drawElbow,
                drawTarget: personalProfile.drawElbowTarget,
                drawDiff: analysis.drawDiff,
                bowArm: currentMetrics.bowArm,
                bowTarget: personalProfile.bowArmTarget,
                bowDiff: analysis.bowDiff,
                headTilt: currentMetrics.headTilt,
                headStatus: currentMetrics.headStatus,
                shoulderSlope: currentMetrics.shoulderSlope,
                shoulderStatus: currentMetrics.shoulderStatus,
                diagnosis: analysis.diagnosis,
                drill: analysis.drill,
                priority: analysis.priority
            });

            document.getElementById('shotCount').innerText = shotNumber;
            alert(`تیر شماره ${shotNumber} ثبت شد.\nتشخیص: ${analysis.diagnosis}\nاولویت: ${analysis.priority}`);
        }

        function exportToCSV() {
            if (sessionLogs.length === 0) {
                alert("هنوز هیچ تیری ثبت نشده است! ابتدا دکمه «ثبت تیر» را بزنید.");
                return;
            }

            let csvContent = "\uFEFF"; // کدگذاری BOM برای سازگاری کامل زبان فارسی در Excel
            csvContent += "شماره تیر,زمان ثبت,زاویه آرنج کشش,هدف شخصی آرنج,انحراف از هدف,زاویه دست کمان,هدف شخصی دست کمان,انحراف از هدف,شاخص انحراف سر,وضعیت سر,شاخص شیب شانه,وضعیت شانه,تشخیص اصلی خطا,تمرین اصلاحی پیشنهادی,سطح اولویت\n";

            sessionLogs.forEach(row => {
                // پاک‌سازی کاما جهت حفظ ساختار ستون‌های CSV
                const cleanDiagnosis = `"${row.diagnosis.replace(/"/g, '""')}"`;
                const cleanDrill = `"${row.drill.replace(/"/g, '""')}"`;

                csvContent += `${row.shot},${row.time},${row.drawElbow},${row.drawTarget},${row.drawDiff},${row.bowArm},${row.bowTarget},${row.bowDiff},${row.headTilt},${row.headStatus},${row.shoulderSlope},${row.shoulderStatus},${cleanDiagnosis},${cleanDrill},${row.priority}\n`;
            });

            const blob = new Blob([csvContent], { type: 'text/csv;charset=utf-8;' });
            const link = document.createElement("a");
            const dateStr = new Date().toISOString().slice(0, 10);
            link.href = URL.createObjectURL(blob);
            link.download = `SilentArrows_Analysis_${dateStr}.csv`;
            link.click();
        }
